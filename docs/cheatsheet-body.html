Reading several values at once

int a, b;  float x;  char op;

scanf("%d %d", &a, &b);
scanf("%f %c %f", &x, &op, &x);
Every variable needs & in scanf — except arrays and strings, which are already addresses.

Reading a char safely

char c;
scanf(" %c", &c);   // note the leading space
The space matters. Without it, %c reads the Enter key left over from the previous scanf and your program looks like it skipped an input.

Skipping an input (assignment suppression)

// input: 101 50 24.50 A
// read the 50 but do not store it
scanf("%d %*d %f %c", &code, &price, &cat);

// input: number: 78 position: 4
// skip the words, keep the numbers
scanf("%*s %d %*s %d", &num, &pos);

The * means "match this, then throw it away". It appears in set 01 and again in the array insert/delete problems.

Operators

set 02

Integer division and modulus

7 / 2      // 3   (int / int throws away the rest)
7 % 2      // 1   (the remainder)
7.0 / 2    // 3.5 (one float makes it float)
(float)7 / 2   // 3.5 (cast to force it)
The single most common mistake. Averages, percentages and division answers come out wrong because both operands were int. Cast one of them.

Increment: pre vs post

int a = 5;
printf("%d", a++);   // prints 5, then a becomes 6
printf("%d", ++a);   // a becomes 7, then prints 7

x += 3;   x -= 3;
x *= 3;   x /= 3;   x %= 3;

Conditions

set 03

The chain

if (m >= 90)       printf("A\n");
else if (m >= 86)  printf("A-\n");
else if (m >= 82)  printf("B+\n");
else               printf("F\n");
Order matters. Go from the highest boundary down. If you start with m >= 55, everyone gets a D.

Classifying a character

if ((c >= 'a' && c <= 'z') ||
    (c >= 'A' && c <= 'Z'))  printf("Alphabet\n");
else if (c >= '0' && c <= '9') printf("Digit\n");
else                            printf("Special\n");

Leap year

if ((y % 4 == 0 && y % 100 != 0) || y % 400 == 0)
    printf("Yes\n");

Even / odd, positive / negative

n % 2 == 0        // even (works for negatives too)
n > 0             // positive
n == 0            // zero is its own case - check it!

Menu with a submenu

scanf("%d", &choice);
if (choice == 1)      printf("Addition: %g\n", a + b);
else if (choice == 4) {
    if ((int)b == 0) { printf("Error: Divisor is zero\n"); }
    else {
        scanf("%d", &sub);
        if (sub == 1) printf("Quotient: %d\n", (int)a / (int)b);
        else          printf("Remainder: %d\n", (int)a % (int)b);
    }
}
Check the divisor you actually divide by. If you divide by (int)b, then test (int)b == 0. Testing b != 0 lets 0.5 through and the program crashes.

switch (alternative to a menu chain)

switch (choice) {
    case 1: printf("Add\n");  break;
    case 2: printf("Sub\n");  break;
    default: printf("Invalid\n");
}
Missing break falls through into the next case.

Loops

set 04

The three forms

for (i = 1; i <= n; i++) { }

while (i <= n) { i++; }

do { i++; } while (i <= n);   // runs at least once

break and continue

if (g == x) { printf("Right\n"); break; }   // leave the loop
if (s[i] == ' ') continue;                  // skip to next turn

Comma-separated output

for (i = 1; i <= n; i++) {
    printf("%d", i);
    if (i < n) printf(", ");   // no comma after the last
}
printf("\n");

Reading n values and accumulating

float x, sum = 0;
for (i = 0; i < n; i++) {
    scanf("%f", &x);
    sum += x;
}
printf("%.2f\n", sum / n);

Alternating series 1 − 2 + 3 − 4

for (i = 1; i <= n; i++) {
    if (i % 2 == 1) sum += i;
    else            sum -= i;
}

Building 1, 12, 123, 1234

long term = 0;
for (i = 1; i <= n; i++) {
    term = term * 10 + i;
    sum += term;
}

Infinite loop with a sentinel

while (1) {
    scanf(" %c", &c);
    if (c == 'A') break;
    printf("Input %d: %c\n", i++, c);
}

Taylor series for sin(x)

term = x;  sum = x;
for (i = 1; i < 10; i++) {
    term = term * (-1) * x * x / ((2*i) * (2*i + 1));
    sum += term;
}

Each term is built from the one before — no factorial needed.

Arrays

set 05

Declare, read, print

int a[100], n, i;
scanf("%d", &n);
for (i = 0; i < n; i++) scanf("%d", &a[i]);

for (i = 0; i < n; i++) printf("%d ", a[i]);
printf("\n");
Declare it big ([100]) and use only n of it. Indexes run 0 to n-1.

Print in reverse

for (i = n - 1; i >= 0; i--) printf("%d ", a[i]);

Reverse in place

for (i = 0; i < n / 2; i++) {   // only halfway!
    temp       = a[i];
    a[i]       = a[n-1-i];
    a[n-1-i]   = temp;
}
Loop to n/2, not n — going all the way swaps everything back.

Max / min with index

int max = a[0], maxi = 0;
for (i = 1; i < n; i++)
    if (a[i] > max) { max = a[i]; maxi = i; }
Start from a[0], not 0. Starting at 0 breaks on all-negative arrays.

Even values vs even indexes

if (a[i] % 2 == 0) sum += a[i];   // even VALUE

for (i = 0; i < n; i += 2) sum += a[i];   // even INDEX
Read the question twice. These two are asked as separate problems and look almost identical.

Search, reporting every position

int found = 0;
for (i = 0; i < n; i++) if (a[i] == key) found++;

if (found == 0) printf("NOT FOUND\n");
else {
    printf("FOUND at index position: ");
    int first = 1;
    for (i = 0; i < n; i++)
        if (a[i] == key) {
            if (!first) printf(", ");
            printf("%d", i);  first = 0;
        }
}

Insert at a position

for (i = n; i > pos; i--)   // shift RIGHT, from the back
    a[i] = a[i-1];
a[pos] = num;
n++;

Delete at a position

for (i = pos; i < n - 1; i++)   // shift LEFT, from the front
    a[i] = a[i+1];
n--;
Direction is everything. Insert walks backwards, delete walks forwards. Going the wrong way copies one value over the whole array.

Bubble sort (ascending)

for (i = 0; i < n - 1; i++)
    for (j = 0; j < n - 1 - i; j++)
        if (a[j] > a[j+1]) {
            temp = a[j]; a[j] = a[j+1]; a[j+1] = temp;
        }

Flip to < for descending.

Remove duplicates

int b[100], bn = 0;
for (i = 0; i < n; i++) {
    int dup = 0;
    for (j = 0; j < bn; j++) if (b[j] == a[i]) dup = 1;
    if (!dup) b[bn++] = a[i];
}

Set operations — all three share one shape

// INTERSECTION: in A and also in B
for (i = 0; i < n; i++) {
    int inB = 0;
    for (j = 0; j < m; j++) if (a[i] == b[j]) inB = 1;
    if (inB) printf("%d ", a[i]);
}

// DIFFERENCE A-B: in A but NOT in B   -> if (!inB)
// UNION: print all of A, then each B not already in A
Learn one, get three. Inner loop sets a flag, outer loop acts on it. Only the final if changes.

Nested loops and patterns

set 06

The shape of every pattern problem

for (i = 1; i <= n; i++) {      // rows
    for (j = 1; j <= i; j++) {  // columns
        printf("*");
    }
    printf("\n");               // end of row - OUTSIDE inner loop
}
The newline goes in the outer loop. Put it inside the inner loop and you get one character per line.

Right triangle / inverted

// growing: 1 to i
for (j = 1; j <= i; j++)

// shrinking: i to n
for (j = i; j <= n; j++)

Pyramid: spaces then stars

for (i = 1; i <= n; i++) {
    for (j = 1; j <= n - i; j++) printf(" ");   // leading spaces
    for (j = 1; j <= 2*i - 1; j++) printf("*"); // odd count
    printf("\n");
}
Two magic numbers. A centred pattern of n rows is 2*n-1 wide, and row i has 2*i-1 stars with n-i spaces before it.

Numbers instead of stars

printf("%d", j);          // 1 2 3 across
printf("%d", i);          // same number down a row
printf("%d", i + j - 1);  // 123 / 234 / 345

Distance-from-edge trick

int width = 2*n - 1;
for (p = 0; p < width; p++) {
    int left = p + 1, right = width - p;
    int v = (left < right) ? left : right;  // 1,2,3,2,1
    if (v <= i) printf("%d", v);
    else        printf("_");
}

Useful for diamonds and any pattern that mirrors around the centre.

Multiplication table

for (i = 1; i <= n; i++) {
    for (j = 1; j <= 10; j++)
        printf("%d x %d = %d\n", i, j, i*j);
    printf("\n");
}

2D arrays

set 07

Declare, read, print

int a[50][50], r, c, i, j;
scanf("%d %d", &r, &c);

for (i = 0; i < r; i++)
    for (j = 0; j < c; j++)
        scanf("%d", &a[i][j]);

for (i = 0; i < r; i++) {
    for (j = 0; j < c; j++) printf("%d ", a[i][j]);
    printf("\n");           // newline per ROW
}

i is the row, j is the column. Always.

The position conditions

i == j              // main diagonal \
i + j == n - 1      // other diagonal /
i == 0 || i == n-1 ||
j == 0 || j == n-1  // border
i < j               // above the diagonal
i > j               // below the diagonal
Memorise these five. Identity matrix, diagonal sums, triangular printing and border problems are all just one of these inside the standard double loop.

Transpose

for (i = 0; i < r; i++)
    for (j = 0; j < c; j++)
        t[j][i] = a[i][j];      // indexes swapped

Symmetric check

int symmetric = 1;
for (i = 0; i < n; i++)
    for (j = 0; j < n; j++)
        if (a[i][j] != a[j][i]) symmetric = 0;

Addition

c[i][j] = a[i][j] + b[i][j];   // inside the double loop

Multiplication — three loops

for (i = 0; i < r1; i++)
    for (j = 0; j < c2; j++) {
        c[i][j] = 0;            // reset before summing!
        for (k = 0; k < c1; k++)
            c[i][j] += a[i][k] * b[k][j];
    }
Two traps. Reset c[i][j] = 0 inside the j loop, and remember c1 must equal r2 or the multiplication is undefined.

Row and column sums

for (i = 0; i < r; i++) {
    sum = 0;                    // reset per row
    for (j = 0; j < c; j++) sum += a[i][j];
    printf("Row %d: %d\n", i, sum);
}
// for columns, swap the loops: j outside, i inside

Strings

set 08 — done manually, no string.h

How this lab reads a string

char s[200];
fgets(s, 200, stdin);     // reads the WHOLE line, spaces included

int len = 0;
while (s[len] != '\0' && s[len] != '\n') len++;   // this is strlen()
This while is the key to the whole chapter. fgets keeps the Enter key as '\n' at the end, so every loop must stop at both '\0' and '\n' or you print a stray blank line and count one character too many.

Why not scanf("%s")

scanf("%s", s);        // stops at the first SPACE
fgets(s, 200, stdin);  // takes the whole line

Any problem mentioning a sentence needs fgets.

Walking a string

int i = 0;
while (s[i] != '\0' && s[i] != '\n') {
    printf("%c", s[i]);
    i++;
}

Case conversion by arithmetic

if (c >= 'a' && c <= 'z') c = c - 'a' + 'A';   // to upper
if (c >= 'A' && c <= 'Z') c = c - 'A' + 'a';   // to lower

Characters are numbers: 'A' is 65, 'a' is 97, '0' is 48. So c - '0' turns a digit character into its value.

Counting vowels

if (c=='a'||c=='e'||c=='i'||c=='o'||c=='u'||
    c=='A'||c=='E'||c=='I'||c=='O'||c=='U') count++;

Counting words

int count = 0, inWord = 0;
while (s[i] != '\0' && s[i] != '\n') {
    if (s[i] != ' ') {
        if (inWord == 0) { count++; inWord = 1; }
    } else inWord = 0;
    i++;
}
Do not just count spaces — two spaces in a row would count an extra word. The flag handles it.

Palindrome (two pointers)

int i = 0, j = len - 1, pal = 1;
while (i < j) {
    if (s[i] != s[j]) pal = 0;
    i++;  j--;
}

Reverse a string

for (i = len - 1; i >= 0; i--) printf("%c", s[i]);
printf("\n");

Concatenate two strings

i = 0; while (a[i] != '\0' && a[i] != '\n') printf("%c", a[i++]);
j = 0; while (b[j] != '\0' && b[j] != '\n') printf("%c", b[j++]);
printf("\n");

Most frequent character

for (i = 0; i < len; i++) {
    if (s[i] == ' ') continue;
    count = 0;
    for (j = 0; j < len; j++)
        if (s[j] == s[i]) count++;
    if (count > best) { best = count; bestChar = s[i]; }
}

Compare in one case (convert both) if the question ignores case.

Algorithms worth memorising

these keep coming back

Break a number into digits

while (n != 0) {
    digit = n % 10;   // last digit
    n = n / 10;       // chop it off
}

The base of digit-count, digit-sum, reverse and palindrome problems.

Reverse a number

rev = 0;
while (t != 0) {
    rev = rev * 10 + t % 10;
    t = t / 10;
}
if (rev == n) printf("Palindrome\n");
Copy n first. The loop destroys it, and you need the original to compare against.

Factorial

long f = 1;
for (i = 1; i <= n; i++) f *= i;

Use long and %ld — int overflows past 12!.

Fibonacci

a = 1; b = 1;
for (i = 1; i <= n; i++) {
    printf("%d ", a);
    t = a + b;  a = b;  b = t;
}

Prime check

int isPrime = 1;
if (n < 2) isPrime = 0;
for (i = 2; i < n; i++)
    if (n % i == 0) isPrime = 0;
Handle n < 2 separately. 0 and 1 are not prime but the loop never runs for them, so they would pass.

GCD and LCM

for (i = 1; i <= a && i <= b; i++)
    if (a % i == 0 && b % i == 0) gcd = i;

lcm = a / gcd * b;   // divide BEFORE multiplying
a / gcd * b, not a * b / gcd — dividing first keeps the number small and avoids overflow.

Power without pow()

long p = 1;
for (i = 1; i <= y; i++) p *= x;

nCr

long num = 1, den = 1;
for (i = 1; i <= r; i++) {
    num *= (n - i + 1);
    den *= i;
}
printf("%ld\n", num / den);

Power of two

while (n > 0 && n % 2 == 0) n = n / 2;
if (n == 1) printf("Yes\n");
The n > 0 guard is not optional. Without it, input 0 loops forever because 0/2 is still 0.

Swap two variables

temp = a;  a = b;  b = temp;
